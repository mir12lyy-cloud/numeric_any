440[[nodiscard]] constexpr ::std::optional<T> numeric_cast(const numeric_any& x) noexcept {
441 if constexpr (Policy == casting_policy::strict) {
442 if (!x.can_safe_convert_to<T>()) return ::std::nullopt;
443 } else if constexpr (Policy == casting_policy::normal) {
444 if (x.empty()) return ::std::nullopt;
445 if constexpr (::std::is_floating_point_v<T>) {
446 if (x.width > sizeof(T)) return ::std::nullopt;
447 } else if constexpr (::std::is_unsigned_v<T>) {
448 if (x.float_point || !x.positive) return ::std::nullopt;
449 } else {
450 if (x.float_point || (x.is_unsigned && x.width >= sizeof(T))) return ::std::nullopt;
451 }
452 }
453 switch (x.type) {
454 FOR_EACH_TYPES_TO(FUNCTION_TO_RESTORE_VALUE)
455 case details::numeric_types::EMPTY:
456 default:
457 return ::std::nullopt;
458 }

556template <details::ArithmeticWithExplictSign T>
557constexpr bool numeric_any::can_safe_convert_to() const noexcept {
558 if (empty()) return false;
559 if (is_same_type<T>()) return true; // Safe when the tag of the "T" is same as inner tag.
560 if (type == details::numeric_types::BOOL) {
561 return true; // Bool is safe to cast to all types.
562 }
563 if constexpr (::std::is_floating_point_v<T>) {
564 return float_point && width <= sizeof(T); // Only support promotions in floating numbers.
565 } else if constexpr (::std::is_unsigned_v<T>) {
566 if (is_unsigned) return width <= sizeof(T); // Promotions between two unsigned numbers are safe.
567 if (!float_point) {
568 if (positive) return width <= sizeof(T); // Promotions from non-negative numbers are safe.
569 return false;
570 }
571 return false;
572 } else {
573 // Support to promotions to a wider signed integral type.
574 if (is_unsigned) return width < sizeof(T);
575 // Allowing equal width converting when both signed integral numbers.
576 if (!float_point) return width <= sizeof(T);
577 return false;
578 }
579}

657[[nodiscard]] constexpr ::std::partial_ordering
658numeric_any::operator<=>(details::ArithmeticWithExplictSign auto x) const noexcept {
659 if (empty()) return ::std::partial_ordering::unordered;
660 // The NaN will be checked first.
661 if constexpr (::std::is_floating_point_v<decltype(x)>) {
662 if (x != x) return ::std::partial_ordering::unordered; // NOLINT
663 if constexpr (::std::is_same_v<decltype(x), float>) {
664 if (type == details::numeric_types::FLOAT) return unchecked_numeric_cast<float>(*this) <=> x;
665 if (type == details::numeric_types::DOUBLE) return unchecked_numeric_cast<double>(*this) <=> x;
666 } else if constexpr (::std::is_same_v<decltype(x), double>) {
667 if (type != details::numeric_types::LONG_DOUBLE) return unchecked_numeric_cast<double>(*this) <=> x;
668 }
669 return unchecked_numeric_cast<long double>(*this) <=> x;
670 } else {
671 if (positive != x >= decltype(x){}) return static_cast<int>(positive) <=> static_cast<int>(x >= decltype(x){});
672 switch (type) {
673 case details::numeric_types::UNSIGNED_LONG_LONG:
674 return unchecked_numeric_cast<unsigned long long>(*this) <=> static_cast<unsigned long long>(x);
675 case details::numeric_types::FLOAT:
676 return unchecked_numeric_cast<float>(*this) <=> x;
677 case details::numeric_types::DOUBLE:
678 return unchecked_numeric_cast<double>(*this) <=> x;
679 case details::numeric_types::LONG_DOUBLE:
680 return unchecked_numeric_cast<long double>(*this) <=> x;
681 default:
682 return unchecked_numeric_cast<long long>(*this) <=> x;
683 }
684 }

688[[nodiscard]] constexpr ::std::partial_ordering numeric_any::operator<=>(const numeric_any& x) const noexcept {
689 if (empty() || x.empty()) return ::std::partial_ordering::unordered;
690 if (type == details::numeric_types::LONG_DOUBLE || x.type == details::numeric_types::LONG_DOUBLE)
691 return unchecked_numeric_cast<long double>(*this) <=> unchecked_numeric_cast<long double>(x);
692 if (type == details::numeric_types::DOUBLE || x.type == details::numeric_types::DOUBLE)
693 return unchecked_numeric_cast<double>(*this) <=> unchecked_numeric_cast<double>(x);
694 if (type == details::numeric_types::FLOAT || x.type == details::numeric_types::FLOAT)
695 return unchecked_numeric_cast<float>(*this) <=> unchecked_numeric_cast<float>(x);
696 // The signed will be checked first, so 12U < -1 will be false.
697 if (positive != x.positive) return static_cast<int>(positive) <=> static_cast<int>(x.positive);
698 if (type == details::numeric_types::UNSIGNED_LONG_LONG || x.type == details::numeric_types::UNSIGNED_LONG_LONG)
699 return unchecked_numeric_cast<unsigned long long>(*this) <=> unchecked_numeric_cast<unsigned long long>(x);
700 return unchecked_numeric_cast<long long>(*this) <=> unchecked_numeric_cast<long long>(x);
701}
